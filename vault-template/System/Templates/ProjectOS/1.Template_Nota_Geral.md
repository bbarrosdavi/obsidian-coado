<%* // ============================================================ // 1. SOLICITAÇÃO DE DADOS INICIAIS // ============================================================ // Solicita o Título
const title = await tp.system.prompt("Título da Nota (Nome Descritivo)");
if (!title) return; 

// Tratamento de segurança para o Título
const yamlTitle = title.replace(/"/g, '\\"');

// Solicita o Tipo - ADICIONADO TASK E DOC
const typeDisplay = [
    "✅ Tarefa (TASK)",
    "📄 Documentação (DOC)",
    "🤝 Alinhamento (ALIN)", 
    "📊 Relatório (REL)",
    "🛠️ Técnica (TEC)"
];
const typeValues = ["TASK", "DOC", "ALIN", "REL", "TEC"];

const noteType = await tp.system.suggester(typeDisplay, typeValues);
if (!noteType) return;

// Variáveis que serão preenchidas pelas lógicas abaixo
let projectFolder = "";
let dashboardLink = "";
let projectTag = "";
let areaTag = "";
let typeTag = "";
let specificContent = "";
let yamlStatus = "ativo"; // Default para notas
let yamlPriority = "";    // Apenas para tasks
let yamlDueDate = "";     // Apenas para tasks

// ============================================================ // 2. LÓGICA DE SELEÇÃO (RAMIFICAÇÃO TASK vs GERAL) // ============================================================ 

if (noteType === "TASK") {
    // --- LÓGICA ESPECÍFICA PARA TASKS ---

    // 2.1 Selecionar Área
    const rootChildren = app.vault.getRoot().children;
    const areas = rootChildren
        .filter(x => x instanceof tp.obsidian.TFolder && !x.name.startsWith(".") && !x.name.startsWith("Ω") && x.name !== "Z_Vault")
        .map(x => x.name);
    
    const selectedArea = await tp.system.suggester(areas, areas);
    if (!selectedArea) return;

    // 2.2 Selecionar Projeto dentro da Área
    const areaFolderObj = app.vault.getAbstractFileByPath(selectedArea);
    let projects = [];
    if (areaFolderObj && areaFolderObj.children) {
        projects = areaFolderObj.children
            .filter(x => x instanceof tp.obsidian.TFolder)
            .map(x => x.name);
    }

    const selectedProject = await tp.system.suggester(projects, projects);
    if (!selectedProject) return;

    // 2.3 Definir Caminhos e Tags
    projectFolder = `${selectedArea}/${selectedProject}/01_Gestão`;
    
    // Tenta encontrar o Dashboard do projeto para linkar
    const projFiles = app.vault.getMarkdownFiles().filter(f => f.path.startsWith(`${selectedArea}/${selectedProject}`));
    // Prioriza arquivos que contenham 'DASHBOARD' no nome
    const dashFile = projFiles.find(f => f.name.includes("DASHBOARD")) || projFiles.find(f => f.name.includes(selectedProject) && !f.name.includes("MOC"));
    dashboardLink = dashFile ? `[[${dashFile.basename}]]` : `[[${selectedProject}]]`;

    // Normalização de Tags
    areaTag = `area/${selectedArea.toLowerCase()}`;
    projectTag = `projeto/${selectedProject.toLowerCase().replace(/\s+/g, "-").normalize("NFD").replace(/[̀-ͯ]/g, "")}`;
    typeTag = "tipo/task";
    
    // Frontmatter específico de Task
    yamlStatus = "open";
    yamlPriority = "normal"; // Default
    
    // Conteúdo do Template TASK (Copiado de Template_Task.md)
    specificContent = `### Objetivo

### Entregáveis

**Critérios de Aceitação:**

### Status

### Dúvidas e Decisões

### Próximos passos

---
### Notas

---
### Log de Desenvolvimento`;

} else {
    // --- LÓGICA ORIGINAL PARA OUTROS TIPOS (ALIN, REL, TEC) ---

    // Busca arquivos que tenham a tag 'dashboard' ou comecem com 'DASHBOARD_'
    const allFiles = app.vault.getMarkdownFiles();
    const dashboardFiles = allFiles.filter(file => {
        const cache = app.metadataCache.getFileCache(file);
        const tags = tp.obsidian.getAllTags(cache) || [];
        return tags.some(t => t.includes("dashboard")) || file.name.startsWith("DASHBOARD_");
    });

    // Seleciona o Dashboard
    const selectedDashboard = await tp.system.suggester(
        (f) => `${f.basename} (${f.parent.path})`, 
        dashboardFiles
    );
    if (!selectedDashboard) return;

    projectFolder = selectedDashboard.parent.path;
    dashboardLink = `[[${selectedDashboard.basename}|${selectedDashboard.basename.replace("DASHBOARD_", "").replace(/_/g, " ")}]]`;

    // Extrai tag do projeto original
    const dashboardCache = app.metadataCache.getFileCache(selectedDashboard);
    const dashboardTags = tp.obsidian.getAllTags(dashboardCache) || [];
    const rawProjectTag = dashboardTags.find(t => t.includes("projeto/")) || "projeto/geral";
    projectTag = rawProjectTag.replace("#", ""); 
    
    // Tenta inferir a área da pasta pai se possível, ou usa padrão
    // (Simplificação: não extrai área explicitamente no legado, mantém compatibilidade)
    areaTag = ""; // Tag de área opcional no modo legado ou extraída se necessário

    // Conteúdo Específico
    switch(noteType) {
        case "ALIN":
            typeTag = "tipo/reuniao";
            specificContent = `## Síntese
---

---
## Pontos Discutidos

## Decisões Tomadas

## Action Items

## Próximos Passos

## Links Relacionados`;
            break;

        case "REL":
            typeTag = "tipo/documentacao";
            specificContent = `## Descrição da atividade

## Dificuldades

## Conclusões

## Referências`;
            break;

        case "TEC":
            typeTag = "tipo/nota-tecnica";
            specificContent = `## Tópicos Principais

## Anotações

## Ações e Tarefas

## Referências`;
            break;

        case "DOC":
            typeTag = "tipo/documentacao";
            specificContent = `## Contexto e Objetivo

## Detalhamento

## Histórico e Logs

## Referências`;
            break;
    }
}

// ============================================================ // 3. ORGANIZAÇÃO E CRIAÇÃO DO ARQUIVO // ============================================================ 

const datePrefix = tp.date.now("YYYY-MM-DD");
const safeTitle = title.replace(/[/\\:]/g, "").replace(/\s+/g, "_"); 
const newFileName = `${datePrefix}_${noteType}_${safeTitle}`;

// Renomeia e Move
await tp.file.rename(newFileName);
await tp.file.move(`${projectFolder}/${newFileName}`);

// Constrói lista de Tags para o Frontmatter
let finalTags = [projectTag, typeTag];
if (areaTag) finalTags.push(areaTag);
if (noteType !== "TASK") finalTags.push("status/ativo"); // Tasks usam campo status separado

// ============================================================ // 4. GERAÇÃO DO CONTEÚDO FINAL // ============================================================ 
_%> 
---
title: "<% yamlTitle %>"
tags: [<%- finalTags.join(", ") %>]
created: <% tp.date.now("YYYY-MM-DD HH:mm") %>
modified: <% tp.date.now("YYYY-MM-DD HH:mm") %>
project: "<% dashboardLink %>"
<%* if (noteType === "TASK") { _%>
status: <% yamlStatus %>
priority: <% yamlPriority %>
due_date: <% yamlDueDate %>
<%* } else { _%>
code: "<% noteType %>"
status: <% yamlStatus %>
<%* } _%>
---

# <% title %>

> [!note]+ Navegação
> [[GES_DASHBOARD_CENTRAL|Central]] → <% dashboardLink %>

---

<% specificContent %>

---
> [!note]+ Navegação
> [[GES_DASHBOARD_CENTRAL|Central]] → <% dashboardLink %>
