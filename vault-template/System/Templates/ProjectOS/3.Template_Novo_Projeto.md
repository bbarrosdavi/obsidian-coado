// 6. Criação do Arquivo
const activeFile = app.workspace.getActiveFile();
if (activeFile && (activeFile.name.toLowerCase().includes("untitled") || activeFile.name.toLowerCase().includes("sem título")) && activeFile.stat.size < 100) {
    // Se o template for rodado em cima de uma nota vazia "Untitled", renomeia e move ela
    await tp.file.rename(`DASHBOARD_${safeName}`);
    await tp.file.move(`${rootFolder}/DASHBOARD_${safeName}.md`);
    
    // Como a nota vai ficar aberta com o conteúdo retornado, não precisamos recriá-la
} else {
    // Caso contrário (rodado via atalho sem nota nova aberta), cria o arquivo normalmente
    await tp.file.create_new(content, `DASHBOARD_${safeName}`, true, rootFolder);
}

// 7. Injeta o bloco do Projeto na Dashboard da Área
try {
    const areaDashboardPath = `${area}/DASHBOARD_${area}.md`;
    const areaDashboardFile = app.vault.getAbstractFileByPath(areaDashboardPath);
    
    if (areaDashboardFile) {
        const fileContent = await app.vault.read(areaDashboardFile);
        
        // Tenta achar a linha `> > [!multi-column]` logo abaixo de PROJETOS e injeta na marra baseando-se na estrutura padrão
        const regex = /(> \[!tip\]\+ PROJETOS\s*>\s*> > \[!multi-column\]\s*)/;
        if (regex.test(fileContent)) {
            const blockToInsert = `$1> >\n> > > [!abstract]+ ${projectName}\n> > > **Status**: Planejamento | **Prioridade**: Média\n> > > - [[DASHBOARD_${safeName}|Dashboard]] | [[MOC_${safeName}|MOC]]\n> > > - **Foco**: ${projectDesc || 'Pendente de detalhamento'}\n`;
            const updatedContent = fileContent.replace(regex, blockToInsert);
            await app.vault.modify(areaDashboardFile, updatedContent);
        }
    }
} catch (e) {
    console.log("Erro ao injetar projeto na dashboard da área:", e);
}

// Retorna o conteúdo se renomeamos a nota atual (o templater irá injetar)
if (app.workspace.getActiveFile()?.name === `DASHBOARD_${safeName}.md`) {
    tR += content;
