<%*
// 1. Solicita Dados
const areaName = await tp.system.prompt("Nome da Área (Ex: LAMIA, Financeiro)");
if (!areaName) return;
const areaDesc = await tp.system.prompt("Descrição da Área");
const areaTag = "area/" + areaName.toLowerCase();

// 2. Define Caminho
const rootFolder = areaName;

// 3. Cria Pasta (se não existir)
try {
    await app.vault.createFolder(rootFolder);
} catch (e) {
    console.log("Pasta já existe ou erro na criação.");
}

// 4. Define Conteúdo
const content = `---
title: "${areaName} - Dashboard"
tags: [tipo/dashboard, ${areaTag}, status/ativo]
created: ${tp.date.now("YYYY-MM-DD HH:mm")}
modified: ${tp.date.now("YYYY-MM-DD HH:mm")}
status: "ativo"
project: [[MOC_${areaName}|${areaName}]]
cssclasses: dashboard
---

> [!info]+ Área de ${areaName}
> ${areaDesc}

---

> [!tip]+ PROJETOS
>
> > [!multi-column]

> [!multi-column]
>
> > [!quote]+ Em Andamento (Tasks)
> > \`\`\`dataview
> > TABLE WITHOUT ID
> >   file.link as "Task",
> >   due_date as "Due Date"
> > FROM "${areaName}"
> > WHERE contains(tags, "tipo/task") AND (status = "in-progress" OR status = "open")
> > SORT priority DESC, due_date ASC
> > LIMIT 10
> > \`\`\`
>
> > [!success]+ Completadas Recentes
> > \`\`\`dataview
> > TABLE WITHOUT ID
> >   file.link as "Task",
> >   completedDate as "Concluída"
> > FROM "${areaName}"
> > WHERE contains(tags, "tipo/task") AND status = "done" AND completedDate >= date(today) - dur(3 month)
> > SORT completedDate DESC
> > LIMIT 8
> > \`\`\`

> [!example]- Subtasks Gerais da Área
> \`\`\`tasks
> not done
> path includes ${areaName}
> limit 20
> \`\`\`

---

> [!multi-column]
>
>> [!info]+ Recursos e Wiki
>> - [[${areaName}/Wiki|Wiki Geral]]
>> - **Tipo**: Base de Conhecimento
>
>> [!info]- Dashboards Relacionados
>> - [[GES_DASHBOARD_CENTRAL|Dashboard Central]]
>
>> [!note]- Navegação
> [[GES_DASHBOARD_CENTRAL|Central]] → **${areaName}**
`;

// 5. Cria Arquivo
let isActiveFileUsed = false;
const activeFile = app.workspace.getActiveFile();
if (activeFile && (activeFile.name.toLowerCase().includes("untitled") || activeFile.name.toLowerCase().includes("sem título")) && activeFile.stat.size < 100) {
    await tp.file.rename(`DASHBOARD_${areaName}`);
    await tp.file.move(`${rootFolder}/DASHBOARD_${areaName}.md`);
    isActiveFileUsed = true;
} else {
    await tp.file.create_new(content, `DASHBOARD_${areaName}`, true, rootFolder);
}

const areaMocFileName = `MOC_${areaName}`;
const areaMocContent = `---
title: "${areaName} - Map of Content"
tags: [tipo/moc, status/ativo, ${areaTag}]
created: ${tp.date.now("YYYY-MM-DD HH:mm")}
modified: ${tp.date.now("YYYY-MM-DD HH:mm")}
project: [[DASHBOARD_${areaName}|${areaName}]]
status: ativo
priority: "alta"
cssclasses: dashboard, wide-page
---
# ${areaName} - Map of Content

> [!note]- Navegação
> [[GES_DASHBOARD_CENTRAL|Central]] → [[DASHBOARD_${areaName}|${areaName}]]

> [!info]+ Sobre a Área ${areaName}
> Descrição e organização centralizada para a área **${areaName}**.
---
## Visão Geral da Área

> [!abstract]+ Objetivo
> Objetivo principal da área ${areaName}.
>
> **Fase Atual**: Consolidação
> **Marco**: Organização de Projetos

> [!tip]+ Progresso da Área
> 
> \`\`\`dataview
> TABLE WITHOUT ID
>   length(filter(rows, (r) => r.status = "done")) as "Done",
>   length(filter(rows, (r) => r.status = "in-progress")) as "In Progress",
>   length(filter(rows, (r) => r.status = "open" OR r.status = "none")) as "Open",
>   "<progress value=\\'" + string(length(filter(rows, (r) => r.status = "done"))) + "\\' max=\\'" + string(length(rows)) + "\\'></progress> " + round(length(filter(rows, (r) => r.status = "done")) * 100.0 / length(rows), 1) + "%" as "%"
> FROM "${rootFolder}"
> WHERE contains(tags, "tipo/task") AND contains(tags, "${areaTag}")
> GROUP BY file.name
> LIMIT 1
> \`\`\`

## Projetos e Documentação

> [!multi-column]
>
> > [!info]+ Projetos da Área
> > \`\`\`dataview
> > LIST FROM "${rootFolder}"
> > WHERE contains(tags, "tipo/dashboard") AND contains(tags, "${areaTag}")
> > SORT created DESC
> > LIMIT 5
> > \`\`\`
>
> > [!info]+ Documentos Relevantes
> > \`\`\`dataview
> > LIST FROM "${rootFolder}"
> > WHERE contains(tags, "tipo/documentacao") AND contains(tags, "${areaTag}")
> > SORT created DESC
> > LIMIT 5
> > \`\`\`

> [!info]+ Todos os Documentos (tipo/documentacao)
> \`\`\`dataview
> TABLE title AS "Título", file.folder AS "Pasta", created AS "Criado"
> FROM "${rootFolder}"
> WHERE contains(tags, "tipo/documentacao") AND contains(tags, "${areaTag}")
> SORT created DESC
> \`\`\`

> [!info]+ Todas as Notas (tipo/nota-tecnica)
> \`\`\`dataview
> TABLE title AS "Título", file.folder AS "Pasta", created AS "Criado"
> FROM "${rootFolder}"
> WHERE contains(tags, "tipo/nota-tecnica") AND contains(tags, "${areaTag}")
> SORT created DESC
> \`\`\`
`;

await tp.file.create_new(areaMocContent, areaMocFileName, true, rootFolder);

// 6. Injeta o bloco da Área na Dashboard Central
try {
    const centralDashboardPath = `GES_DASHBOARD_CENTRAL.md`;
    const centralDashboardFile = app.vault.getAbstractFileByPath(centralDashboardPath);
    
    if (centralDashboardFile) {
        const fileContent = await app.vault.read(centralDashboardFile);
        
        const regex = /(> \[!tip\]\+ ÁREAS PRINCIPAIS\s*>\s*> > \[!multi-column\]\s*)/;
        if (regex.test(fileContent)) {
            const blockToInsert = `$1> >\n> > > [!abstract]+ ${areaName}\n> > > **Foco**: ${areaDesc || 'Pendente de detalhamento'}\n> > > - [[DASHBOARD_${areaName}|Dashboard]] | [[MOC_${areaName}|MOC]]\n`;
            const updatedContent = fileContent.replace(regex, blockToInsert);
            await app.vault.modify(centralDashboardFile, updatedContent);
        }
    }
} catch (e) {
    console.log("Erro ao injetar área na dashboard central:", e);
}

// Retorna o conteúdo se renomeamos a nota atual (o templater irá injetar na nota limpa)
if (isActiveFileUsed) {
    tR += content;
}

return "";
_%>
